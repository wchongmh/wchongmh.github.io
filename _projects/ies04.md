  <!-- Conclusion -->
  <section class="project-section conclusion-section">
    <h2 class="section-header">
      <span class="section-number">10</span>
      Conclusion & Recommendations
    </h2>
    
    <div class="conclusion-content">
      <div class="findings">
        <h3>Key Findings</h3>
        <ul>
          <li>Most living spaces meet or exceed recommended daylight factors</li>
          <li>Some task areas require supplemental lighting</li>
          <li>Transition spaces need improved illumination for safety</li>
          <li>Automatic controls can significantly improve energy efficiency</li>
        </ul>
      </div>
      
      <div class="final-recommendations">
        <h3>Final Recommendations</h3>
        <ol>
          <li>Implement light shelves in north-facing rooms to improve daylight distribution</li>
          <li>Install occupancy sensors in low-usage areas like cloakrooms</li>
          <li>Use high-efficiency T8 lamps throughout the property</li>
          <li>Provide task lighting in kitchen work areas</li>
          <li>Improve stair and hallway lighting with automatic controls</li>
        </ol>
      </div>
